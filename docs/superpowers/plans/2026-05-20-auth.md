# Auth Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Implement the authentication subsystem — secure token storage via `keyring`, transparent JWT access-token refresh, and a full OIDC browser-based login flow using ORCID as the identity provider via OneDep.

**Architecture:** Two classes with a clean split of responsibilities: `TokenStore` owns the token lifecycle (read/write keyring, detect expiry, refresh transparently) and `OidcAuth` owns the OIDC flow (open browser, run loopback HTTP server, exchange one-time code for tokens, hand off to `TokenStore`). Both depend only on `DepositConfig` and the `exceptions` module from Plan 1 — no dependency on the session or API layers.

**Tech Stack:** Python 3.9+, `keyring` (new dep), `pyjwt` (already a dep), `requests` (already a dep), `pytest`, `pytest-httpserver` (already added in Plan 1).

**Prerequisite:** Plan 1 complete (`exceptions.py` and `config.py` must exist).

## Task 3: `TokenStore`

`TokenStore` uses `keyring` to persist the JWT access token and opaque refresh token. It transparently refreshes the access token 60 seconds before expiry by posting to the OneDep refresh endpoint.

**Keyring layout:**
- Service name: `"nextdep_dsp"`
- Username for access token: `"{hostname}:access_token"`
- Username for refresh token: `"{hostname}:refresh_token"`

Using the hostname as part of the key allows separate tokens per OneDep server (test vs production).

# src/nextdep_dsp/auth/token.py
from __future__ import annotations

import time

import jwt
import keyring
import keyring.errors
import requests

from nextdep_dsp.exceptions import AuthError


class TokenStore:
    _SERVICE = "nextdep_dsp"

    def __init__(self, hostname: str) -> None:
        self._hostname = hostname

    def _key(self, kind: str) -> str:
        return f"{self._hostname}:{kind}"

    def store_tokens(self, access_token: str, refresh_token: str) -> None:
        keyring.set_password(self._SERVICE, self._key("access_token"), access_token)
        keyring.set_password(self._SERVICE, self._key("refresh_token"), refresh_token)

    def get_access_token(self, refresh_url: str, ssl_verify: bool = True) -> str:
        token = keyring.get_password(self._SERVICE, self._key("access_token"))
        if token is None:
            raise AuthError(
                "No access token stored. Run OidcAuth.login() first."
            )
        if self._is_expired(token):
            token = self._refresh(refresh_url, ssl_verify)
        return token

    def clear_tokens(self) -> None:
        for kind in ("access_token", "refresh_token"):
            try:
                keyring.delete_password(self._SERVICE, self._key(kind))
            except keyring.errors.PasswordDeleteError:
                pass

    def _is_expired(self, token: str) -> bool:
        try:
            decoded = jwt.decode(token, options={"verify_signature": False})
            return int(decoded.get("exp", 0)) <= int(time.time()) + 60
        except jwt.DecodeError:
            return True

    def _refresh(self, refresh_url: str, ssl_verify: bool) -> str:
        refresh_token = keyring.get_password(
            self._SERVICE, self._key("refresh_token")
        )
        if refresh_token is None:
            raise AuthError(
                "No refresh token stored. Run OidcAuth.login() first."
            )
        try:
            response = requests.post(
                refresh_url,
                json={"grant_type": "refresh_token", "refresh_token": refresh_token},
                verify=ssl_verify,
                timeout=30,
            )
            response.raise_for_status()
        except requests.RequestException as exc:
            raise AuthError(f"Token refresh failed: {exc}") from exc

        data = response.json()
        new_access = data["access_token"]
        new_refresh = data.get("refresh_token", refresh_token)
        self.store_tokens(new_access, new_refresh)
        return new_access
```

## Task 4: `OidcAuth`

`OidcAuth` implements the browser-based OIDC login flow described in `docs/auth_flow.md`. It:
1. Generates a random `state` nonce for CSRF protection
2. Opens the system browser to `{hostname}/auth/oauth/login?state=STATE`
3. Starts a temporary HTTP server on `127.0.0.1:PORT` to receive the redirect callback
4. Waits (up to 120 s) for the callback containing the one-time code
5. `POST`s the one-time code to `{hostname}/api/v1/auth/token` to receive `access_token` + `refresh_token`
6. Stores both tokens in `TokenStore`

# src/nextdep_dsp/auth/oidc.py
from __future__ import annotations

import secrets
import threading
import webbrowser
from http.server import BaseHTTPRequestHandler, HTTPServer
from urllib.parse import parse_qs, urlparse

import requests

from nextdep_dsp.auth.token import TokenStore
from nextdep_dsp.config import DepositConfig
from nextdep_dsp.exceptions import AuthError


class OidcAuth:
    def __init__(
        self,
        config: DepositConfig,
        token_store: TokenStore,
        port: int = 8765,
        timeout: int = 120,
    ) -> None:
        self._config = config
        self._token_store = token_store
        self._port = port
        self._timeout = timeout

    def login(self) -> None:
        state = secrets.token_urlsafe(16)
        login_url = f"{self._config.hostname}/auth/oauth/login?state={state}"

        received_code: list[str] = []
        server_done = threading.Event()

        class _Handler(BaseHTTPRequestHandler):
            def do_GET(self):
                params = parse_qs(urlparse(self.path).query)
                code = params.get("code", [None])[0]
                if code:
                    received_code.append(code)
                self.send_response(200)
                self.end_headers()
                self.wfile.write(
                    b"Authentication complete. You may close this tab."
                )
                threading.Thread(
                    target=self.server.shutdown, daemon=True
                ).start()
                server_done.set()

            def log_message(self, *args: object) -> None:
                pass

        server = HTTPServer(("127.0.0.1", self._port), _Handler)
        server_thread = threading.Thread(target=server.serve_forever, daemon=True)
        server_thread.start()

        webbrowser.open(login_url)
        server_done.wait(timeout=self._timeout)
        server_thread.join(timeout=2)

        if not received_code:
            raise AuthError(
                "Authentication timed out or was cancelled. "
                "Please run login() again."
            )

        self._exchange_code(received_code[0])

    def logout(self) -> None:
        self._token_store.clear_tokens()

    def _exchange_code(self, code: str) -> None:
        token_url = f"{self._config.hostname}/api/v1/auth/token"
        try:
            response = requests.post(
                token_url,
                json={"code": code},
                verify=self._config.ssl_verify,
                timeout=30,
            )
            response.raise_for_status()
        except requests.RequestException as exc:
            raise AuthError(f"Code exchange failed: {exc}") from exc

        data = response.json()
        self._token_store.store_tokens(
            access_token=data["access_token"],
            refresh_token=data["refresh_token"],
        )
```
