# Auth Subsystem Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Implement the OAuth2 / OIDC auth flow for nextdep_dsp v2. After a user completes a browser-based OIDC login, OneDep returns a JWT access token and an opaque refresh token. This plan stores them in a plain JSON file (similar to how sessions are stored), manages token lifecycle (expiry + refresh), and provides the `OidcAuth` class that drives the browser login and loopback callback server.

**Prerequisite:** Plan 1 (`2026-05-20-core-library.md`) must be complete — specifically `exceptions.py` and `config.py`.

**New dependency:** `pyjwt` — for decoding JWTs to check expiry without signature verification.

Token storage: `~/.config/nextdep/tokens.json`. Each hostname gets its own section, keyed by a normalised form of the hostname. This is a runtime state file (not user-edited config), kept alongside `config.toml` in the same directory for consistency.

## Task 3 — `auth/token.py`: `TokenStore`

`TokenStore` persists JWT access tokens and opaque refresh tokens as JSON at
`~/.config/nextdep/tokens.json`. Multiple server hostnames co-exist in the same
file, each under a normalised hostname key.

**Token file format** (`~/.config/nextdep/tokens.json`):
```json
{
  "deposit_wwpdb_org": {
    "access_token": "eyJ...",
    "refresh_token": "opaque-string"
  },
  "deposit_example_com_8080": {
    "access_token": "eyJ...",
    "refresh_token": "opaque-string"
  }
}
```

Hostname normalisation: strip `https://` / `http://`, replace all non-alphanumeric characters with `_`, strip leading/trailing `_`.

All writes use atomic replace (write temp file → `os.replace`) to avoid corruption on crash.

# src/nextdep_dsp/auth/token.py
from __future__ import annotations

import json
import os
import re
import time
from pathlib import Path
from typing import Union

import jwt as pyjwt
import requests

from nextdep_dsp.exceptions import AuthError

_DEFAULT_TOKEN_FILE = Path.home() / ".config" / "nextdep" / "tokens.json"


class TokenStore:
    """Persists JWT access tokens and opaque refresh tokens per hostname.

    Tokens are stored in ~/.config/nextdep/tokens.json (or a path supplied
    in tests). Multiple hostnames co-exist under normalised keys. All writes
    use atomic replace (write-then-rename) to prevent corruption on crash.
    """

    def __init__(
        self,
        hostname: str,
        token_file: Path | None = None,
        ssl_verify: bool = True,
    ) -> None:
        self._hostname = hostname
        self._token_file = token_file or _DEFAULT_TOKEN_FILE
        self._ssl_verify = ssl_verify
        self._token_file.parent.mkdir(parents=True, exist_ok=True)

    def _key(self) -> str:
        h = re.sub(r"^https?://", "", self._hostname)
        h = re.sub(r"[^a-zA-Z0-9]", "_", h)
        return h.strip("_")

    def _read(self) -> dict:
        if not self._token_file.exists():
            return {}
        with self._token_file.open() as f:
            try:
                return json.load(f)
            except (json.JSONDecodeError, ValueError):
                return {}

    def _write(self, data: dict) -> None:
        tmp = self._token_file.with_suffix(".json.tmp")
        try:
            with tmp.open("w") as f:
                json.dump(data, f, indent=2)
            os.replace(tmp, self._token_file)
        except Exception:
            tmp.unlink(missing_ok=True)
            raise

    def store_tokens(self, access_token: str, refresh_token: str) -> None:
        """Write access and refresh tokens for this hostname."""
        data = self._read()
        data[self._key()] = {
            "access_token": access_token,
            "refresh_token": refresh_token,
        }
        self._write(data)

    def get_access_token(self, refresh_url: str, ssl_verify: bool | None = None) -> str:
        """Return a valid access token, refreshing automatically if expired.

        Args:
            refresh_url: Full URL of the token refresh endpoint.
            ssl_verify: Override SSL verification; falls back to constructor value.

        Raises:
            AuthError: If no token is stored or refresh fails.
        """
        verify = ssl_verify if ssl_verify is not None else self._ssl_verify
        data = self._read()
        section = data.get(self._key(), {})
        token = section.get("access_token")
        if not token:
            raise AuthError("No access token stored. Run OidcAuth.login() first.")
        if self._is_expired(token):
            refresh_token = section.get("refresh_token")
            if not refresh_token:
                raise AuthError("No refresh token stored. Run OidcAuth.login() first.")
            token = self._refresh(refresh_url, refresh_token, verify)
        return token

    def clear_tokens(self) -> None:
        """Remove all tokens for this hostname from the token file."""
        data = self._read()
        data.pop(self._key(), None)
        self._write(data)

    def _is_expired(self, token: str) -> bool:
        try:
            payload = pyjwt.decode(
                token,
                options={"verify_signature": False},
                algorithms=["RS256", "HS256", "none"],
            )
            return payload.get("exp", 0) < time.time() + 60
        except Exception:
            return True

    def _refresh(self, refresh_url: str, refresh_token: str, ssl_verify: bool) -> str:
        try:
            resp = requests.post(
                refresh_url,
                json={"grant_type": "refresh_token", "refresh_token": refresh_token},
                verify=ssl_verify,
                timeout=30,
            )
            resp.raise_for_status()
            body = resp.json()
        except Exception as exc:
            raise AuthError(f"Token refresh failed: {exc}") from exc

        new_access = body.get("access_token")
        if not new_access:
            raise AuthError("Token refresh response missing access_token")
        new_refresh = body.get("refresh_token", refresh_token)

        data = self._read()
        data[self._key()] = {"access_token": new_access, "refresh_token": new_refresh}
        self._write(data)
        return new_access
```

## Task 4 — `auth/oidc.py`: `OidcAuth`

`OidcAuth` drives the full browser-based OIDC flow (RFC 8252 loopback redirect):
1. Generate a random `state` nonce.
2. Open the system browser to `{hostname}/auth/oauth/login?state={state}`.
3. Start a plain HTTP server on `127.0.0.1:{port}`.
4. Wait up to `timeout` seconds for `GET /callback?code=ONE_TIME_CODE&state=STATE`.
5. Validate the returned `state` matches the generated nonce.
6. POST the code to `{hostname}/api/v1/auth/token` to exchange for tokens.
7. Call `token_store.store_tokens(access_token, refresh_token)`.

# src/nextdep_dsp/auth/oidc.py
from __future__ import annotations

import secrets
import threading
import webbrowser
from http.server import BaseHTTPRequestHandler, HTTPServer
from urllib.parse import parse_qs, urlparse

import requests

from nextdep_dsp.auth.token import TokenStore
from nextdep_dsp.config import DepositConfig
from nextdep_dsp.exceptions import AuthError


class OidcAuth:
    """Drives the RFC 8252 loopback OIDC flow for OneDep.

    Usage:
        auth = OidcAuth(config, token_store)
        auth.login()   # opens browser, waits for callback, stores tokens
        auth.logout()  # clears stored tokens
    """

    def __init__(
        self,
        config: DepositConfig,
        token_store: TokenStore,
        port: int = 8765,
        timeout: int = 120,
    ) -> None:
        self._config = config
        self._token_store = token_store
        self._port = port
        self._timeout = timeout
        self._state = secrets.token_urlsafe(32)
        self._result: dict | None = None
        self._error: str | None = None

    def login(self) -> None:
        """Launch OIDC flow: open browser → loopback callback → exchange code → store tokens.

        Raises:
            AuthError: On timeout, state mismatch, or token exchange failure.
        """
        self._state = secrets.token_urlsafe(32)
        self._result = None
        self._error = None

        login_url = (
            f"{self._config.hostname}/auth/oauth/login"
            f"?state={self._state}"
            f"&redirect_uri=http://127.0.0.1:{self._port}/callback"
        )
        webbrowser.open(login_url)

        self._run_callback_server()

        if self._error:
            raise AuthError(self._error)
        if self._result is None:
            raise AuthError(
                f"OIDC login timeout: no callback received within {self._timeout}s"
            )

        self._exchange_code(self._result["code"])

    def logout(self) -> None:
        """Clear stored tokens for this hostname."""
        self._token_store.clear_tokens()

    def _run_callback_server(self) -> None:
        auth = self

        class _Handler(BaseHTTPRequestHandler):
            def do_GET(self):
                parsed = urlparse(self.path)
                if parsed.path != "/callback":
                    self.send_response(404)
                    self.end_headers()
                    return

                params = parse_qs(parsed.query)
                code = (params.get("code") or [None])[0]
                state = (params.get("state") or [None])[0]

                if state != auth._state:
                    auth._error = (
                        f"OIDC state mismatch: expected {auth._state!r}, got {state!r}"
                    )
                    self.send_response(400)
                    self.end_headers()
                    self.wfile.write(b"State mismatch. You may close this window.")
                else:
                    auth._result = {"code": code}
                    self.send_response(200)
                    self.end_headers()
                    self.wfile.write(b"Login successful. You may close this window.")

            def log_message(self, format, *args):
                pass  # suppress access log

        server = HTTPServer(("127.0.0.1", self._port), _Handler)
        server.timeout = self._timeout

        def _serve():
            server.handle_request()
            server.server_close()

        thread = threading.Thread(target=_serve, daemon=True)
        thread.start()
        thread.join(timeout=self._timeout + 1)

        if thread.is_alive():
            server.server_close()
            raise AuthError(
                f"OIDC login timeout: no callback received within {self._timeout}s"
            )

    def _exchange_code(self, code: str) -> None:
        token_url = f"{self._config.hostname}/api/v1/auth/token"
        try:
            resp = requests.post(
                token_url,
                json={"code": code},
                verify=self._config.ssl_verify,
                timeout=30,
            )
            resp.raise_for_status()
            body = resp.json()
        except Exception as exc:
            raise AuthError(f"Token exchange failed: {exc}") from exc

        access_token = body.get("access_token")
        refresh_token = body.get("refresh_token")
        if not access_token or not refresh_token:
            raise AuthError(
                "Token exchange response missing access_token or refresh_token"
            )
        self._token_store.store_tokens(access_token, refresh_token)
```
